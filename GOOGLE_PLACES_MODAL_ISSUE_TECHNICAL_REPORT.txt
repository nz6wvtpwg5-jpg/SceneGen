```typescript
import { useLoadScript, Autocomplete } from '@react-google-maps/api';

export function PlacesAutocomplete({ value, onChange, placeholder, label, error }) {
  const { isLoaded, loadError } = useLoadScript({
    googleMapsApiKey: process.env.NEXT_PUBLIC_GOOGLE_MAPS_API_KEY,
    libraries: ["places"],
  });

  const [autocomplete, setAutocomplete] = useState<google.maps.places.Autocomplete | null>(null);
  const [inputValue, setInputValue] = useState(value);

  const onLoad = (autocompleteInstance: google.maps.places.Autocomplete) => {
    console.log('🟢 [PlacesAutocomplete/maps] onLoad called');
    setAutocomplete(autocompleteInstance);
  };

  const onPlaceChanged = () => {
    console.log('🔥 [PlacesAutocomplete/maps] onPlaceChanged called!');
    if (autocomplete) {
      const place = autocomplete.getPlace();
      // Extract address, lat, lng, city, state
      onChange(address, lat, lng, name, city, state);
    }
  };

  return (
    <Autocomplete onLoad={onLoad} onPlaceChanged={onPlaceChanged}>
      <Input
        type="text"
        placeholder={placeholder}
        value={inputValue}
        onChange={(e) => setInputValue(e.target.value)}
      />
    </Autocomplete>
  );
}
```

SECTION 6: ROOT CAUSE ANALYSIS
================================================================================

THE FUNDAMENTAL PROBLEM:

Google Places Autocomplete relies on a specific event sequence:
1. User clicks dropdown item (.pac-item)
2. Google's code internally fires 'place_changed' event on Autocomplete instance
3. Our onPlaceChanged handler receives the event
4. We extract place data

IN A DIALOG MODAL:
1. User clicks dropdown item (.pac-item)
2. Dialog's focus management intercepts the event
3. Dialog refocuses to modal content
4. Input blur event fires
5. Google's 'place_changed' event is LOST or CANCELLED
6. Our onPlaceChanged never fires

EVIDENCE:
- Blur event fires immediately on dropdown click
- isSelectingRef.current is always false (mousedown never captured)
- getPlace() in blur handler returns incomplete data: {name: 'atla'}
- No 'place_changed' event in console logs
- onChange callback never fires

THE MYSTERY:
Why does @react-google-maps/api Autocomplete's onPlaceChanged not fire when
inside a Radix UI Dialog? The component works perfectly outside the Dialog.

POSSIBLE CAUSES:
1. Dialog's focus trap cancels bubbling events from external elements
2. Dialog's event listeners preventDefault on click events
3. React portal event handling differs from normal DOM
4. Google's code can't find the Autocomplete instance in the portal context
5. Z-index/stacking context issues prevent events from reaching .pac-container
