      const entity = req.activeEntity;
      if (entity.trialExpiresAt && now > entity.trialExpiresAt) {
        return res.status(403).json({
          error: 'Trial expired. Contact support to continue.'
        });
      }
      next();

    function computeTourDates(tourId) {
      const shows = await db.query(
        'SELECT MIN(showDate) as start, MAX(showDate) as end
         FROM shows WHERE tourId = ? AND status != "cancelled"',
        [tourId]
      );
      await db.query(
        'UPDATE tours SET startDate = ?, endDate = ? WHERE id = ?',
        [shows.start, shows.end, tourId]
      );
    }
